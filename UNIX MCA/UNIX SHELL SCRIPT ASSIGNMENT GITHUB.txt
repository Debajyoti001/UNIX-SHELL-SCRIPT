QN 1 Greet User According to System Time

#!/bin/bash

hour=$(date +%H)

if [ $hour -ge 5 ] && [ $hour -lt 12 ]
then
    echo "GOOD MORNING"
elif [ $hour -ge 12 ] && [ $hour -lt 17 ]
then
    echo "GOOD AFTERNOON"
elif [ $hour -ge 17 ] && [ $hour -lt 21 ]
then
    echo "GOOD EVENING"
else
    echo "GOOD NIGHT"
fi

QN 12. lower.sh

#!/bin/bash

for file in "$1"/*
do
    base=$(basename "$file")
    lower=$(echo "$base" | tr 'A-Z' 'a-z')

    if [ -e "$1/$lower" ]
    then
        echo "Warning: Not overwriting $lower"
    else
        mv "$file" "$1/$lower"
    fi
done

QN14. Factorial Using Function

#!/bin/bash

fact()
{
    n=$1
    f=1

    while [ $n -gt 1 ]
    do
        f=$((f*n))
        n=$((n-1))
    done

    echo $f
}

fact $1
